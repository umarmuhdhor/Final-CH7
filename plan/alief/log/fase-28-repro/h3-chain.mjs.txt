// ---------- the island side, through the REAL files (#486) ----------
// live-island.js is a plain content script: it reads `document`, `location`, `browser` and `addEventListener` when it
// runs. They exist only inside withPage(), so X-08's block.js / blocked-page.js still see no DOM (they check for one).
// The message it sends goes through live.js show() like background.js `case "liveScore"` (with #535 it forwards
// `label` and `secondWarning`; live.js keeps only known values). Without #535, live-island.js sends neither.
const island = await (async () => {
    const sent = [];
    const pageGlobals = () => {
        class Document {}
        Object.defineProperty(Document.prototype, "visibilityState", { get: () => "visible" });
        return { Document, document: new Document(), location: { hostname: "bca-promo.example" }, browser: { runtime: { sendMessage: async (m) => void sent.push(m) } }, addEventListener: () => {} };
    };
    const page = pageGlobals();
    const withPage = async (fn) => { Object.assign(globalThis, page); try { return await fn(); } finally { for (const k of Object.keys(page)) delete globalThis[k]; } };
    let listener = null;
    const subscribe = P.subscribe;
    P.subscribe = (fn) => ((listener = fn), () => {});
    try { await withPage(() => import(`${R}/live-island.js`)); } finally { P.subscribe = subscribe; }
    return {
        // pipeline state -> live-island.js message -> live.js show() -> the POST /live body
        async body(state) {
            sent.length = 0;
            await withPage(() => listener(state));
            const m = sent.at(-1);
            if (!m) throw new Error("live-island.js sent nothing for this state");
            let body = null;
            const live = createLive({ getSession: async () => ({ aiBaseURL: "https://ai.test", apiBaseURL: "https://api.test", token: "t" }), post: async (_s, b) => (body = b) });
            await live.show({ domain: "bca-promo.example", score: m.score, status: m.status, analyzing: m.analyzing, label: m.label, secondWarning: m.secondWarning });
            return body;
        },
        // backend live.ts contentState() copies label / secondWarning as is (#486); ScoreActivityAttributes.swift `tone` +
        // `headline`, ScoreLiveActivity.swift `statusText`: a label only replaces "Aman", danger starts at secondWarning ?? 50.
        tone: (b) => (b.status === "blocked" ? "blocked" : b.status === "warning" ? (b.score >= (b.secondWarning ?? 50) ? "danger" : "warning") : "safe"),
        text(b) {
            const tone = this.tone(b);
            if (b.analyzing) return "Menganalisis…";
            if (tone === "safe" && b.label === "marked_safe") return "Ditandai aman oleh Guardian";
            if (tone === "safe" && b.label === "offline_check") return "Cek online tidak tersedia";
            return { safe: "Aman", warning: "Hati-hati", danger: "Risiko tinggi", blocked: "Diblokir" }[tone];
        },
    };
})();
