import XCTest
import UIKit

/// Fase 28 hunt (hunter 4): UI edge-case matrix. Not a gate: it never fails a cell, it writes a screenshot
/// (JPEG) and the accessibility tree (JSON) per cell to `TEST_RUNNER_F28_DIR/<TEST_RUNNER_F28_TAG>/`.
/// Config from the environment: `F28_LANG` (id|en), `F28_SIZE` (UICTContentSizeCategory…), `F28_THEME`
/// (light|dark), `F28_EXTRA` (comma list: bold, contrast). Checks run afterwards in `h4-check.py`.
final class Fase28MatrixUITests: XCTestCase {
    private var env: [String: String] { ProcessInfo.processInfo.environment }
    private var lang: String { env["F28_LANG"] ?? "id" }
    private var dir: URL {
        let url = URL(fileURLWithPath: env["F28_DIR"] ?? NSTemporaryDirectory()).appendingPathComponent(env["F28_TAG"] ?? "default")
        try? FileManager.default.createDirectory(at: url, withIntermediateDirectories: true)
        return url
    }
    private var app = XCUIApplication()
    private var notes: [String] = []

    override func setUp() {
        continueAfterFailure = true
    }

    // MARK: Launch + dump

    private func guardian(_ scenario: String = "figma", family: Bool = true) -> [String] {
        ["-settings.mode", "guardian", "-settings.familyModeOn", family ? "YES" : "NO", "-mock.scenario", scenario]
    }

    private func protected(_ scenario: String = "figma") -> [String] {
        ["-settings.mode", "protected", "-mock.scenario", scenario]
    }

    @MainActor
    @discardableResult
    private func launch(_ arguments: [String], onboarded: Bool = true, extensionActive: Bool = true, notifications: String = "granted",
                        settle: TimeInterval = 1.8) -> XCUIApplication {
        app = XCUIApplication()
        app.terminate()
        notes = []
        var args = [
            "-UITestMock", "YES", "-UITestSkipSplash", "YES", "-UITestReset", "YES", "-UITestStillLoops", "YES",
            "-settings.language", lang, "-AppleLanguages", "(\(lang))", "-AppleLocale", lang == "id" ? "id_ID" : "en_US",
            "-settings.theme", env["F28_THEME"] ?? "light",
            "-UITestNotifications", notifications,
        ]
        if let size = env["F28_SIZE"], !size.isEmpty { args += ["-UIPreferredContentSizeCategoryName", size] }
        let extra = (env["F28_EXTRA"] ?? "").split(separator: ",").map(String.init)
        if extra.contains("bold") { args += ["-settings.boldText", "YES"] }
        if extra.contains("contrast") { args += ["-settings.highContrast", "YES"] }
        if onboarded { args += ["-settings.onboardingCompleted", "YES", "-settings.onboardingConnectPassed", "YES"] }
        if extensionActive { args += ["-UITestExtensionActive", "YES"] }
        app.launchArguments = args + arguments
        app.launchEnvironment["TZ"] = "Asia/Jakarta"
        app.launch()
        Thread.sleep(forTimeInterval: settle)
        // GD-01 "Notifikasi mati" popup covers Home once per launch (it shows with `granted` too when the
        // launch arguments are read after the first check); close it unless the cell is about it.
        if notifications != "denied", app.alerts.firstMatch.exists {
            let button = app.alerts.buttons.firstMatch
            notes.append("closed alert: \(app.alerts.firstMatch.label)")
            if button.exists { button.tap() }
            Thread.sleep(forTimeInterval: 0.8)
        }
        return app
    }

    @MainActor
    private func dump(_ name: String) {
        let screenshot = XCUIScreen.main.screenshot()
        if let jpeg = screenshot.image.jpegData(compressionQuality: 0.55) {
            try? jpeg.write(to: dir.appendingPathComponent("\(name).jpg"))
        }
        var nodes: [[String: Any]] = []
        func walk(_ snapshot: XCUIElementSnapshot, depth: Int) {
            let frame = snapshot.frame
            var node: [String: Any] = [
                "t": snapshot.elementType.rawValue, "d": depth,
                "x": Double(frame.minX), "y": Double(frame.minY), "w": Double(frame.width), "h": Double(frame.height),
                "en": snapshot.isEnabled, "sel": snapshot.isSelected,
            ]
            if !snapshot.identifier.isEmpty { node["id"] = snapshot.identifier }
            if !snapshot.label.isEmpty { node["l"] = snapshot.label }
            if let value = snapshot.value { node["v"] = "\(value)" }
            nodes.append(node)
            for child in snapshot.children { walk(child, depth: depth + 1) }
        }
        if let snapshot = try? app.snapshot() { walk(snapshot, depth: 0) } else { notes.append("snapshot failed") }
        let payload: [String: Any] = ["name": name, "notes": notes, "nodes": nodes]
        if let data = try? JSONSerialization.data(withJSONObject: payload) {
            try? data.write(to: dir.appendingPathComponent("\(name).json"))
        }
    }

    @MainActor
    private func element(_ id: String) -> XCUIElement { app.descendants(matching: .any)[id].firstMatch }

    /// Taps the element when it is there (scrolling up to `scrolls` times to reach it). A miss is a note in the dump.
    @MainActor
    @discardableResult
    private func tap(_ id: String, scrolls: Int = 0, wait: TimeInterval = 3) -> Bool {
        let target = element(id)
        _ = target.waitForExistence(timeout: wait)
        var left = scrolls
        while left > 0, !(target.exists && target.isHittable) {
            app.swipeUp(velocity: .slow)
            left -= 1
        }
        guard target.exists else {
            notes.append("missing: \(id)")
            return false
        }
        if target.isHittable { target.tap() } else {
            notes.append("not hittable, tapped frame: \(id)")
            target.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: 0.5)).tap()
        }
        Thread.sleep(forTimeInterval: 1.0)
        return true
    }

    @MainActor
    private func scrollEnd(_ times: Int = 3) {
        for _ in 0..<times { app.swipeUp(velocity: .fast) }
        Thread.sleep(forTimeInterval: 0.8)
    }

    @MainActor
    private func scrollPage() {
        let start = app.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: 0.75))
        let end = app.coordinate(withNormalizedOffset: CGVector(dx: 0.5, dy: 0.30))
        start.press(forDuration: 0.05, thenDragTo: end, withVelocity: .slow, thenHoldForDuration: 0.3)
        Thread.sleep(forTimeInterval: 0.6)
    }

    @MainActor
    private func openLink(_ link: String) {
        app.open(URL(string: link)!)
        let board = XCUIApplication(bundleIdentifier: "com.apple.springboard")
        for title in ["Open", "Buka"] where board.buttons[title].waitForExistence(timeout: 1.5) {
            board.buttons[title].tap()
            break
        }
        Thread.sleep(forTimeInterval: 1.5)
    }

    // MARK: Helpers

    @MainActor
    private func detail(_ name: String, _ scanId: String, _ arguments: [String], pages: Int = 2) {
        launch(arguments + ["-debug.detail", scanId, "-debug.replicaFixture", "YES"], settle: 2.5)
        dump(name)
        if pages >= 1 {
            scrollPage()
            dump("\(name)-mid")
        }
        if pages >= 2 {
            scrollEnd()
            dump("\(name)-end")
        }
    }

    @MainActor
    private func connect(code: String) -> Bool {
        launch(protected("unpaired") + ["-debug.tab", "shield"])
        openLink("shoo://join?code=\(code)")
        return app.textFields["connect.code"].waitForExistence(timeout: 6)
    }
