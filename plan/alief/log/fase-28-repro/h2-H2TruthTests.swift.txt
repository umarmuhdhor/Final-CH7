/// A repository whose calls a test can replace one by one; everything else goes to the fixtures.
final class H2Repo: ShooRepository, @unchecked Sendable {
    let base: MockRepository
    var onMe: (() async throws -> DeviceProfile)?
    var onScans: ((ScanOwnerFilter, ScanStatus?, String?) async throws -> ScanPage)?
    var onScan: ((String) async throws -> ScanResult)?
    var onVerdict: ((String, Verdict.Value) async throws -> Verdict)?
    var onMembers: (() async throws -> [FamilyMember])?
    var onCode: ((Bool) async throws -> FamilyCode)?

    init(mode: UserMode = .guardian) { base = MockRepository(mode: mode, latency: .zero) }

    func registerDevice(name: String, mode: UserMode, avatarId: String?, apnsToken: String?) async throws -> DeviceRegistration {
        try await base.registerDevice(name: name, mode: mode, avatarId: avatarId, apnsToken: apnsToken)
    }
    func me() async throws -> DeviceProfile { if let onMe { return try await onMe() }; return try await base.me() }
    func updateMe(_ update: DeviceUpdate) async throws { try await base.updateMe(update) }
    func createFamilyCode(refresh: Bool) async throws -> FamilyCode {
        if let onCode { return try await onCode(refresh) }
        return try await base.createFamilyCode(refresh: refresh)
    }
    func joinFamily(code: String) async throws -> GuardianInfo { try await base.joinFamily(code: code) }
    func familyMembers() async throws -> [FamilyMember] {
        if let onMembers { return try await onMembers() }
        return try await base.familyMembers()
    }
    func removeMember(deviceId: String) async throws { try await base.removeMember(deviceId: deviceId) }
    func setMemberNotify(deviceId: String, notify: Bool) async throws { try await base.setMemberNotify(deviceId: deviceId, notify: notify) }
    func remindMember(deviceId: String) async throws { try await base.remindMember(deviceId: deviceId) }
    func leaveGuardian() async throws { try await base.leaveGuardian() }
    func becomeProtected(code: String) async throws -> ProtectedSwitch { try await base.becomeProtected(code: code) }
    func deleteMe() async throws { try await base.deleteMe() }
    func scans(owner: ScanOwnerFilter, status: ScanStatus?, cursor: String?) async throws -> ScanPage {
        if let onScans { return try await onScans(owner, status, cursor) }
        return try await base.scans(owner: owner, status: status, cursor: cursor)
    }
    func scan(id: String) async throws -> ScanResult { if let onScan { return try await onScan(id) }; return try await base.scan(id: id) }
    func setVerdict(scanId: String, value: Verdict.Value) async throws -> Verdict {
        if let onVerdict { return try await onVerdict(scanId, value) }
        return try await base.setVerdict(scanId: scanId, value: value)
    }
}

    /// A model says "what you see may be old" through some Bool: `refreshFailed`, `isStale`, `loadFailed` …
    /// (`ActivityModel.refreshFailed` is the existing one, D-alief-A12). Any true one counts.
    static func saysItMayBeOld(_ model: Any) -> Bool {
        Mirror(reflecting: model).children.contains { child in
            let name = (child.label ?? "").lowercased()
            let hit = ["refreshfailed", "stale", "outdated", "reloadfailed", "loadfailed"].contains { name.contains($0) }
            return hit && (child.value as? Bool) == true
        }
    }

    @Test func cell_daysAndCounts() async throws {
        let utc = TimeZone(identifier: "UTC")!
        var jakarta = Calendar(identifier: .gregorian); jakarta.timeZone = TimeZone(identifier: "Asia/Jakarta")!
        var utcCal = Calendar(identifier: .gregorian); utcCal.timeZone = utc
        // 2026-10-21T17:30Z = 22 Oct 00:30 WIB. Registered 2026-10-20T16:59Z = 20 Oct 23:59 WIB.
        let now = Date(timeIntervalSince1970: 1_792_603_800)
        let model = ProfileModel(repository: H2Repo(), settings: Self.settings(), now: { now })
        let since = Date(timeIntervalSince1970: 1_792_515_540)
        print("H2 cell days | WIB \(model.days(since: since, calendar: jakarta)) (expected 3) | UTC \(model.days(since: since, calendar: utcCal)) (expected 2)")
        #expect(model.days(since: since, calendar: jakarta) == 3)
        #expect(model.days(since: since, calendar: utcCal) == 2)
        // Registered "tomorrow" (clock skew) still counts as day 1, never 0 or negative.
        #expect(model.days(since: now.addingTimeInterval(86_400 * 2), calendar: jakarta) == 1)

        // Guardian strip: sums of the members' counters.
        await model.load()
        let stats = model.stats(language: .english)
        let members = model.members
        let blocked = members.reduce(0) { $0 + ($1.stats?.blocked ?? 0) }
        let warnings = members.reduce(0) { $0 + ($1.stats?.warnings ?? 0) }
        print("H2 cell stats | members \(members.count) | blocked \(blocked) warnings \(warnings) | strip \(stats.map(\.value))")
        #expect(stats.count == 3 && stats[1].value == "\(blocked)" && stats[2].value == "\(warnings)")
    }

    @Test func cell_dayGroupingAndTimes() throws {
        let jakarta = TimeZone(identifier: "Asia/Jakarta")!
        var calendar = Calendar(identifier: .gregorian); calendar.timeZone = jakarta
        // 16:59:59Z = 23:59:59 WIB (20 Oct), 17:00:00Z = 00:00:00 WIB (21 Oct).
        let before = Date(timeIntervalSince1970: 1_792_515_599), after = Date(timeIntervalSince1970: 1_792_515_600)
        #expect(calendar.startOfDay(for: before) != calendar.startOfDay(for: after))
        let now = after.addingTimeInterval(3600)
        for language in [AppLanguage.english, .indonesian] {
            let a = Formatters.dayAndTime(before, now: now, language: language, timeZone: jakarta)
            let b = Formatters.dayAndTime(after, now: now, language: language, timeZone: jakarta)
            let future = Formatters.dayAndTime(now.addingTimeInterval(86_400), now: now, language: language, timeZone: jakarta)
            print("H2 cell time | \(language) | 23:59:59 WIB -> \(a) | 00:00:00 WIB -> \(b) | tomorrow -> \(future)")
            #expect(a != b)
            #expect(a.lowercased().contains(language == .english ? "yesterday" : "kemarin"))
            #expect(b.lowercased().contains(language == .english ? "today" : "hari ini"))
        }
        // "Just now" covers a verdict a little in the future (server clock ahead of the phone).
        let verdict = Verdict(value: .safe, byDeviceId: "dev_me", byName: "Adhella", decidedAt: now.addingTimeInterval(240))
        let byline = VerdictNote.byline(for: verdict, viewerDeviceId: "dev_me", now: now, language: .english)
        print("H2 cell time | verdict +240 s | \(byline)")
        #expect(byline.contains("Just now"))
    }
