describe("blocklist truth and paging", () => {
  const get = async (d: Created, since?: string | null) =>
    (await (await call(`/blocklist${since ? `?since=${encodeURIComponent(since)}` : ""}`, { token: d.token })).json()) as { domains: { domain: string; addedAt: string }[]; removed?: { domain: string; removedAt: string }[]; nextSince: string | null };

  it("empty list: nextSince null; with since: echoes it", async () => {
    const p = await createDevice("Ibu");
    expect(await get(p)).toEqual({ domains: [], nextSince: null });
    expect(await get(p, "2026-01-01T00:00:00.000000Z")).toEqual({ domains: [], removed: [], nextSince: "2026-01-01T00:00:00.000000Z" });
  });

  it("queued, rejected and removed domains are never served as blocked; a removed one is told to clients with a cursor", async () => {
    const p = await createDevice("Ibu");
    await admin`insert into blocklist_entries (domain, approved_at, approved_by, removed_at, rejected_at, rejected_by) values
      ('queued.example', null, null, null, null, null),
      ('rejected.example', null, null, null, '2026-11-01T00:00:00Z', 'alief'),
      ('live.example', '2026-11-01T00:00:00Z', 'alief', null, null, null),
      ('removed.example', '2026-11-01T00:00:01Z', 'alief', '2026-11-02T00:00:00Z', null, null)`;
    const full = await get(p);
    expect(full.domains.map((d) => d.domain)).toEqual(["live.example"]);
    expect(full.nextSince).toBe("2026-11-01T00:00:00.000000Z");
    const inc = await get(p, full.nextSince);
    expect(inc.domains).toEqual([]);
    expect(inc.removed!.map((d) => d.domain)).toEqual(["removed.example"]);
    // Boundary: `since` equal to an event time is exclusive; 1 µs earlier includes it.
    expect((await get(p, "2026-11-02T00:00:00.000000Z")).removed).toEqual([]);
    expect((await get(p, "2026-11-01T23:59:59.999999Z")).removed).toHaveLength(1);
    // Re-approved after removal: back in domains with a new time, out of removed.
    await admin`update blocklist_entries set removed_at = null where domain = 'removed.example'`;
    const again = await get(p, inc.nextSince);
    expect([again.domains.map((d) => d.domain), again.removed]).toEqual([["removed.example"], []]);
  });

  it.each([[999, 1], [1000, 1], [1001, 1], [2500, 400], [2500, 1]])("%d approved entries, ties of %d: a client that follows nextSince gets every domain exactly once", async (n, tie) => {
    const p = await createDevice("Ibu");
    await admin`insert into blocklist_entries (domain, approved_at, approved_by)
      select 'd' || lpad(i::text, 5, '0') || '.example', timestamptz '2026-11-01T00:00:00Z' + make_interval(secs => (i / ${tie})), 'alief'
      from generate_series(1, ${n}) i`;
    const seen: string[] = [];
    let since: string | null = null, pages = 0;
    for (;;) {
      const page = await get(p, since);
      pages++;
      if (page.domains.length === 0 && (page.removed ?? []).length === 0) break;
      seen.push(...page.domains.map((d) => d.domain));
      since = page.nextSince;
      if (pages > 20) break;
    }
    expect(seen.length).toBe(n);
    expect(new Set(seen).size).toBe(n);
  });
});

describe("config, code and join page truth", () => {
  it("GET /config/public equals admin.public_config; POST /family/code expiry is 30 min and the invite link carries the code", async () => {
    const g = await createDevice("Gita", "guardian");
    const cfg = (await (await call("/config/public", { token: g.token })).json()) as { warnAt: number; blockAt: number; version: number };
    const row = (await admin`select warn_at, block_at, version from admin.public_config`)[0]!;
    expect([cfg.warnAt, cfg.blockAt, cfg.version]).toEqual([row.warn_at, row.block_at, row.version]);
    const c = (await (await call("/family/code", { method: "POST", token: g.token })).json()) as { code: string; expiresAt: string; inviteUrl: string };
    const db = (await sql`select code, extract(epoch from expires_at - created_at)::int as ttl from family_codes`)[0]!;
    expect([c.code, db.ttl, c.inviteUrl.endsWith(`/j/${c.code}`)]).toEqual([db.code, 1800, true]);
    expect(Math.abs(Date.parse(c.expiresAt) - Date.now() - 1800_000)).toBeLessThan(5000);
  });

  it("GET /j/<code>: a browser navigation tells a live code from a dead one (bounded by the IP budget), markup never reflects the path", async () => {
    const g = await createDevice("Gita", "guardian");
    const c = (await (await call("/family/code", { method: "POST", token: g.token })).json()) as { code: string };
    const nav = { "sec-fetch-mode": "navigate", "sec-fetch-dest": "document" };
    const live = await (await call(`/j/${c.code}`, { headers: nav, ip: "203.0.113.50" })).text();
    const dead = await (await call(`/j/${c.code === "000001" ? "000002" : "000001"}`, { headers: nav, ip: "203.0.113.50" })).text();
    expect([live.includes(`shoo://join?code=${c.code}`), dead.includes("shoo://join")]).toEqual([true, false]);
    let told = 0;
    for (let i = 0; i < 25; i++) {
      const code = String(100000 + i);
      if (code === c.code) continue;
      const html = await (await call(`/j/${code}`, { headers: nav, ip: "203.0.113.51" })).text();
      if (!html.includes("shoo://join")) told++;
    }
    console.log("H1X join-page oracle answers before the IP budget ends:", told);
    expect(told).toBe(20);
    const xss = await (await call(`/j/%3Cscript%3Ealert(1)%3C%2Fscript%3E`, { headers: nav })).text();
    expect(xss).not.toContain("alert(1)");
  });
});
