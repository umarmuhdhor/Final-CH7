-- heavy owner: most scans in 14 days. avg owner: the median. guardian: the family with most scans.
select owner_id as heavy from scans group by 1 order by count(*) desc limit 1 \gset
select owner_id as avg_owner from (select owner_id, count(*) n from scans group by 1) c order by n offset 3000 limit 1 \gset
select l.guardian_id as guardian from family_links l join scans s on s.owner_id = l.protected_id group by 1 order by count(*) desc limit 1 \gset
select protected_id as member from family_links where guardian_id = :'guardian' order by protected_id limit 1 \gset
select guardian_id as avg_guardian from (select l.guardian_id, count(*) n from family_links l join scans s on s.owner_id = l.protected_id group by 1) c order by n offset 750 limit 1 \gset
select protected_id as avg_member from family_links where guardian_id = :'avg_guardian' order by protected_id limit 1 \gset
select encode(token_hash, 'hex') as tokhash from devices where id = :'heavy' \gset
select domain as heavy_domain, count(*) from scans where owner_id = :'heavy' group by 1 order by 2 desc limit 1 \gset
select id as heavy_scan, url as heavy_url from scans where owner_id = :'heavy' and domain = :'heavy_domain' order by created_at desc limit 1 \gset
select id as member_scan, url as member_url, domain as member_domain from scans where owner_id = :'member' order by detected_at desc limit 1 \gset
select s.id as replica_scan, s.owner_id as replica_owner, s.domain as replica_domain from scans s where replica_bytes is not null and created_at > now() - interval '20 hours' order by created_at limit 1 \gset
select v.owner_id as v_owner, v.url as v_url, v.domain as v_domain from verdicts v where v.owner_id = :'member' limit 1 \gset
select code as live_code, guardian_id as code_guardian from family_codes where expires_at > now() limit 1 \gset
select encode(key_hash, 'hex') as rl_dev from rate_limit_hits where bucket = 'scans_create' and hit_at > now() - interval '1 hour' group by 1 order by count(*) desc limit 1 \gset
select encode(sha256('scans_create_all:all'::bytea), 'hex') as rl_all \gset
select to_char(detected_at at time zone 'UTC', 'YYYY-MM-DD"T"HH24:MI:SS.US"Z"') as cur_at, id as cur_id
  from scans where owner_id in (select :'guardian' union select protected_id from family_links where guardian_id = :'guardian')
  order by detected_at desc, id desc offset 180 limit 1 \gset
select string_agg(quote_literal(id), ',') as fam_ids from (select :'guardian'::text as id union select protected_id from family_links where guardian_id = :'guardian') f \gset

\echo ===== S06 recordScan: site state, heavy owner + its busiest domain (db/repo/scans.ts)
:X select count(*) > 0 as seen, coalesce(bool_or(status = 'blocked'), false) as blocked,
          (array_agg(id order by created_at desc, id desc) filter (where public.verdict_link(url) = public.verdict_link(:'heavy_url')))[1] as "linkId"
   from scans where owner_id = :'heavy' and domain = :'heavy_domain';
\echo ===== S07 recordScan: repeat visit (update kept row + device counters)
begin; :X with kept as (
     update scans set detected_at = greatest(detected_at, now()), rdap_registered_at = coalesce(rdap_registered_at, '2026-10-01'::date),
       url = case when 90 > score then :'heavy_url' else url end, page_title = case when 90 > score then 'x' else page_title end,
       signals = case when 90 > score then '[]'::jsonb else signals end, status = case when 90 > score then 'blocked' else status end,
       score = greatest(score, 90)
     where id = :'heavy_scan' returning id)
   update devices d set blocked_total = d.blocked_total + 1, warning_total = d.warning_total + 0 where d.id = :'heavy' and exists (select 1 from kept); rollback;
\echo ===== S08 insertScan: new link (insert + triggers admin_scan_intel + device counters): existing popular domain

     insert into scans (owner_id, url, domain, page_title, favicon_url, score, status, signals, detected_at, thresholds_version, rdap_registered_at)
     select :'heavy', 'https://' || :'heavy_domain' || '/new/page', :'heavy_domain', 'Title', null, 92, 'blocked', signals, now(), 0, '2026-10-01'::date
       from scans where id = :'heavy_scan'
     returning id, owner_id, status),
   counted as (update devices d set blocked_total = d.blocked_total + (ins.status = 'blocked')::int, warning_total = d.warning_total + (ins.status = 'warning')::int from ins where d.id = ins.owner_id)
   select id from ins; rollback;

     returning id, owner_id, status),
   counted as (update devices d set blocked_total = d.blocked_total + 1 from ins where d.id = ins.owner_id)

\echo ===== S13 precheckReplica: reuse lookup + owner/family/global counts of 24 h (one statement)
:X select (select r.replica_path from scans r where r.owner_id = :'replica_owner' and r.domain = :'replica_domain' and r.replica_bytes is not null
            and r.replica_ready_at is not null and r.created_at > now() - interval '24 hours'
            and public.verdict_link(r.url) = (select public.verdict_link(s.url) from scans s where s.id = :'replica_scan') order by r.created_at desc limit 1) as reuse, c.*
   from (with root as (select coalesce((select guardian_id from family_links where protected_id = :'replica_owner'), :'replica_owner') as id),
              members as (select id from root union select protected_id from family_links where guardian_id = (select id from root))
         select count(*) filter (where s.owner_id = :'replica_owner')::int as mine,
                count(*) filter (where s.owner_id in (select id from members))::int as family,
                coalesce(sum(s.replica_bytes) filter (where s.owner_id in (select id from members)), 0)::float8 as "familyBytes",
                count(*)::int as "all", coalesce(sum(s.replica_bytes), 0)::float8 as "allBytes"
         from scans s where s.replica_bytes is not null and s.created_at > now() - interval '24 hours') c;
\echo ===== S15 reserveFragments (count 24 h + update, one statement)
begin; :X with u as (
      select count(*) filter (where owner_id = :'heavy')::int as n_owner, coalesce(sum(fragments_bytes) filter (where owner_id = :'heavy'), 0)::int as b_owner,
             count(*)::int as n_all, coalesce(sum(fragments_bytes), 0)::bigint as b_all
        from scans where fragments_bytes is not null and created_at > now() - interval '24 hours'),
    upd as (update scans s set fragments_bytes = 9000 from u where s.id = :'heavy_scan' and s.owner_id = :'heavy' and s.fragments_bytes is null
              and u.n_owner < 100000 and u.n_all < 100000000 returning s.id)
    select (select count(*) from upd)::int as done, exists (select 1 from scans where id = :'heavy_scan' and owner_id = :'heavy') as present,
           exists (select 1 from scans where id = :'heavy_scan' and owner_id = :'heavy' and fragments_bytes is null) as empty, u.n_owner, u.b_owner, u.n_all, u.b_all::float8 from u; rollback;

-- ---------- cron 5 * * * * notifications ----------
\echo ===== N01 notifications: clear off marks (hourly)
begin; :X update family_links l set off_notified_at = null from devices m where m.id = l.protected_id and l.off_notified_at is not null and m.ext_last_seen_at > now() - interval '24 hours'; rollback;
\echo ===== N02 notifications: protection-off candidates with quiet hours in SQL (limit 40)
:X select l.protected_id from family_links l join devices m on m.id = l.protected_id join devices g on g.id = l.guardian_id
   where l.off_notified_at is null and m.ext_last_seen_at <= now() - interval '24 hours' and g.notify_protection_off and g.apns_token is not null
     and not (g.quiet_hours and (select g.quiet_start = g.quiet_end or (g.quiet_start < g.quiet_end and x.m >= g.quiet_start and x.m < g.quiet_end)
                or (g.quiet_start > g.quiet_end and (x.m >= g.quiet_start or x.m < g.quiet_end))
          from (select (extract(hour from t.local) * 60 + extract(minute from t.local))::int as m
                from (select now() at time zone coalesce((select jsonb_object_agg(lower(z.name), z.name) from pg_timezone_names z) ->> lower(g.time_zone), 'Asia/Jakarta') as local offset 0) t) x))
   order by m.ext_last_seen_at, l.protected_id limit 40;
\echo ===== N03 notifications: weekly summary candidates (no LIMIT: every opted-in guardian, every hour)
:X select id, quiet_hours, quiet_start, quiet_end, time_zone, summary_sent_on from devices
   where weekly_summary and mode = 'guardian' and apns_token is not null and (summary_sent_on is null or summary_sent_on < (now() at time zone 'UTC')::date - 1);

\echo ===== N04 notifications: claim 40 summaries + count 7 days of blocked/warning scans per guardian (2 count subqueries each)
begin; :X update devices d set summary_sent_on = v.day
   from unnest(string_to_array(:'sum_ids', ','), string_to_array(:'sum_days', ',')::date[]) as v(id, day)
   where d.id = v.id and d.weekly_summary and d.mode = 'guardian' and d.apns_token is not null and d.summary_sent_on is distinct from v.day
   returning d.id,
     (select count(*)::int from scans s where s.status = 'blocked' and s.detected_at > now() - interval '7 days' and (s.owner_id = d.id or s.owner_id in (select protected_id from family_links where guardian_id = d.id))) as blocked,
     (select count(*)::int from scans s where s.status = 'warning' and s.detected_at > now() - interval '7 days' and (s.owner_id = d.id or s.owner_id in (select protected_id from family_links where guardian_id = d.id))) as warnings; rollback;
